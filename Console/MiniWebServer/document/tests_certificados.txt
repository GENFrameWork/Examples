# Clave privada EC (PEM)
"C:\Program Files\OpenSSL-Win64\bin\openssl" ecparam -name prime256v1 -genkey -noout -out ecdsa_p256_key.pem

# Certificado autofirmado a partir de esa clave
"C:\Program Files\OpenSSL-Win64\bin\openssl" req -new -x509 -key ecdsa_p256_key.pem -out ecdsa_p256_cert.pem -days 365 -subj "/CN=localhost"

# Certificado en DER "en crudo" -- este es el formato que espera WebServer_PathCertificate
"C:\Program Files\OpenSSL-Win64\bin\openssl" x509 -in ecdsa_p256_cert.pem -outform DER -out ecdsa_p256_cert.der
